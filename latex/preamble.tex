    \def\explain@tmp{\begin{#1}}\else\def\explain@tmp{\par\begin{#1}}\fi\explain@tmp}
\@for\explain@env:={align,align*,alignat,alignat*,gather,gather*,equation,equation*,multline,multline*,flalign,flalign*}\do{%
  \expandafter\let\csname explain@disp@\explain@env\endcsname\@empty}
\makeatother
